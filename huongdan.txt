Bước 10: Tính Cosine Similarity

PHP:

function cosineSimilarity($a, $b)

{

   $dot = 0;

   $normA = 0;

   $normB = 0;



   foreach ($a as $i => $value) {

       $dot += $value * $b[$i];

       $normA += $value * $value;

       $normB += $b[$i] * $b[$i];

   }



   return $dot / (sqrt($normA) * sqrt($normB));

}
